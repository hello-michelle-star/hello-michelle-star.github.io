set.seed(123)

# Select features (expand as needed)
features <- c("model_year", "mileage", "cylinders", "horsepower", "liters", 
              "brand", "fuel_type", "accident", "transmission_type","ext_col")

# Create model matrix (handles categorical variables)
X <- model.matrix(price ~ ., data = train[, c("price", features)])[, -1]

``` r
### Best Subset Selection (BSS)
library(leaps)
bss_fit <- regsubsets(price ~ model_year + mileage + cylinders + horsepower + liters + 
                      brand + fuel_type + accident + transmission_type,
                    data = train, nvmax = 10, method = "exhaustive")
bss_summary <- summary(bss_fit)

# Plot BIC vs number of variables
plot(bss_summary$bic, type = "b", pch = 19, 
     xlab = "Number of Predictors", ylab = "BIC",
     main = "Best Subset Selection")
points(which.min(bss_summary$bic), min(bss_summary$bic), col = "red", pch = 19)
```
