<script>
document.addEventListener('DOMContentLoaded', ()=>{
  const base = '/assets/img/Sampling%20Project/Slide';
  const count = 15;
  const slides = Array.from({length: count}, (_,i)=>({
    src: `${base}${i+1}.jpg`,
    title: `Slide ${i+1}`
  }));

  const grid = document.getElementById('slides-grid');
  grid.innerHTML = slides.map((s,i)=>`
  <figure class="card" data-idx="${i}" tabindex="0" aria-label="${s.title}">
    <img class="thumb" src="${s.src}" alt="${s.title}" loading="lazy" decoding="async">
    <figcaption style="padding:8px 10px;font-size:.95rem;color:#333">
      <a href="#" class="fig-link" data-open="${i}">${s.title}</a>
    </figcaption>
  </figure>
`).join('');


  const dlg = document.getElementById('viewer');
  const vimg = dlg.querySelector('.viewer-img');
  const vtitle = dlg.querySelector('.viewer-title');
  const prevBtn = dlg.querySelector('.prev');
  const nextBtn = dlg.querySelector('.next');
  const closeBtn = dlg.querySelector('.close');
  const nodes = Array.from(document.querySelectorAll('.view-btn'));

  function show(idx){
    idx = Math.max(0, Math.min(idx, slides.length-1));
    dlg.dataset.idx = idx;
    const s = slides[idx];
    vimg.src = s.src; vimg.alt = s.title;
    vtitle.textContent = `${s.title} (${idx+1}/${slides.length})`;
    prevBtn.disabled = (idx===0); nextBtn.disabled = (idx===slides.length-1);
  }

  // open viewer by index
function openAt(i){ show(i); dlg.showModal(); dlg.scrollTop = 0; }

// Click anywhere on a card OR on the caption link
grid.addEventListener('click', (e)=>{
  const linkTarget = e.target.closest('[data-open]');
  if (linkTarget) { e.preventDefault(); openAt(+linkTarget.dataset.open); return; }

  const card = e.target.closest('.card');
  if (card) openAt(+card.dataset.idx);
});

// Keyboard: Enter/Space on the card focuses/open
grid.addEventListener('keydown', (e)=>{
  const card = e.target.closest('.card');
  if (!card) return;
  if (e.key === ' ') { e.preventDefault(); openAt(+card.dataset.idx); }
});



  prevBtn.addEventListener('click', ()=>show(+dlg.dataset.idx-1));
  nextBtn.addEventListener('click', ()=>show(+dlg.dataset.idx+1));
  closeBtn.addEventListener('click', ()=>dlg.close());
  vimg.addEventListener('click', ()=>{ const i=+dlg.dataset.idx; if(i<slides.length-1) show(i+1); });

  window.addEventListener('keydown', e=>{
    if(!dlg.open) return;
    if(e.key==='ArrowRight'){ e.preventDefault(); nextBtn.click(); }
    if(e.key==='ArrowLeft'){ e.preventDefault(); prevBtn.click(); }
    if(e.key==='Escape'){ e.preventDefault(); dlg.close(); }
  });

  dlg.addEventListener('click', e=>{
    const r = dlg.getBoundingClientRect();
    const inside = e.clientX>=r.left && e.clientX<=r.right && e.clientY>=r.top && e.clientY<=r.bottom;
    if(!inside) dlg.close();
  });
});
</script>
